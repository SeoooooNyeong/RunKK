// 예: 상태 전이를 메소드로만 허용하는 엔티티 (JoinRequest)
public class JoinRequest {
    private final long crewId;
    private final long memberId;
    private final LocalDateTime requestedAt;
    private JoinRequestStatus status;
    private LocalDateTime decidedAt;

    JoinRequest(long crewId, long memberId, LocalDateTime now) {
        this.crewId = crewId;
        this.memberId = memberId;
        this.requestedAt = now;
        this.status = JoinRequestStatus.PENDING;
    }

    public void approve(LocalDateTime now) {
        requirePending();
        this.status = JoinRequestStatus.APPROVED;
        this.decidedAt = now;
    }

    public boolean expire(LocalDateTime now) {
        if (status != JoinRequestStatus.PENDING || requestedAt.plusDays(7).isAfter(now)) return false;
        this.status = JoinRequestStatus.EXPIRED;
        this.decidedAt = now;
        return true;
    }

    private void requirePending() {
        if (status != JoinRequestStatus.PENDING) throw new IllegalStateException("처리된 신청");
    }
}

### 13.2 앱·워치 (Kotlin)
